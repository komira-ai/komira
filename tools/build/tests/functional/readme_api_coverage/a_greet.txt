struct Greeter(Writable):
    """A greeter. Its docstring is not read:

    def not_a_method(self):
    """

    comptime KIND = 1
    var name: String

    def __init__(out self, name: String):
        self.name = name

    def hello(self) -> String:
        return "hello " + self.name

    def hello(self, other: String) -> String:
        return "hello " + other

    @staticmethod
    def make() -> Self:
        return Self("made")

    def wave(self):
        pass

    def write_to(self, mut writer: Some[Writer]):
        writer.write(self.name)

    def _secret(self):
        pass


def greet(name: String) -> String:
    return "hello, " + name


def farewell(name: String) -> String:
    return "bye, " + name
