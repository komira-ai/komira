@fieldwise_init
struct Square(Copyable):
    var side: Int

    def area(self) -> Int:
        return self.side * self.side

    def perimeter(self) -> Int:
        return 4 * self.side


# Not exported: its method is not counted.
struct Circle:
    def radius(self) -> Int:
        return 1
